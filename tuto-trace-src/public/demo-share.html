Moyenne de notes

notes = [12, 15, 9, 18]
total = 0
for n in notes:
    total = total + n
moyenne = total / len(notes)
if moyenne >= 12:
    print("Réussi :", moyenne)
else:
    print("Échec :", moyenne)
